# Trackwalk — trackwalk.racing
#
# Policy: do not train on this site; do cite and list it.
#
# Those are different crawlers, and conflating them is the usual mistake. Blocking a training
# crawler costs nothing in traffic. Blocking a search crawler makes the site disappear from the
# answers that send people here. So the training crawlers are blocked by name, the search and
# citation crawlers are welcomed by name, and anything unrecognised is allowed.

# ─── Search & citation crawlers: explicitly welcome ──────────────────────────────────────────
# These drive referral traffic and citations. Do not add any of these to the block list below.
# Note GPTBot is NOT in this group — it is training-only and never the bot that cites you;
# ChatGPT Search citations come from OAI-SearchBot.
User-agent: Googlebot
User-agent: Bingbot
User-agent: DuckDuckBot
User-agent: OAI-SearchBot
User-agent: ChatGPT-User
User-agent: Claude-User
User-agent: Claude-SearchBot
User-agent: PerplexityBot
User-agent: Perplexity-User
Disallow: /public/results.json
Allow: /

# ─── Model-training crawlers: blocked ────────────────────────────────────────────────────────
# Google-Extended is not a crawler and has no user-agent of its own — it is a usage token
# applied to what Googlebot already fetched. Blocking it opts out of Gemini/Vertex training and
# does NOT remove the site from Google Search or AI Overviews, which run on Googlebot's index.
#
# CCBot (Common Crawl) matters more than it looks: its archive feeds a large share of public
# training corpora, so leaving it open undoes most of the rest of this list.
User-agent: GPTBot
User-agent: ClaudeBot
User-agent: anthropic-ai
User-agent: Google-Extended
User-agent: CCBot
User-agent: Applebot-Extended
User-agent: meta-externalagent
User-agent: Bytespider
User-agent: cohere-ai
User-agent: Diffbot
User-agent: Omgilibot
Disallow: /

# ─── Everything else ─────────────────────────────────────────────────────────────────────────
# /public/ must stay crawlable: every word of real content on this site — the feed, 18 seasons
# of results, the rider list — is fetched from /public/*.json at runtime. It was disallowed
# until 2026-09-15, which left Googlebot rendering a 583-character "Loading race feed" shell
# and the results database invisible to search.
#
# results.json is the one exception: the 9 MB canonical artifact the fetchers merge into. The
# app never requests it (it reads the per-season shards in /public/results/), so there is
# nothing to render from it and no reason to spend crawl budget on it. It is listed BEFORE the
# blanket Allow on purpose — Google resolves conflicts by longest match, but order-based
# parsers take the first rule that matches, and with "Allow: /" first those would let it through.
#
# Not listed: xAI / Grok. It publishes no robots.txt contract and its retrieval traffic is
# reported as rotating residential IPs with spoofed browser user-agents, identifying as neither
# "Grok" nor "xAI" — so there is no token here that would reliably do anything.
User-agent: *
Disallow: /public/results.json
Allow: /
